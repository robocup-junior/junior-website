{{- /* Year index for the rules archive, built from the rules-league and rules-year shortcodes on the same page, so it never goes out of date. Param: title (heading text). */ -}} {{- $tokens := findRESubmatch `\{\{<\s*rules-(league|year)\s+([^>]*?)\s*>\}\}` .Page.RawContent -}} {{- $leagues := slice -}} {{- $cur := dict -}} {{- range $tokens -}} {{- $attrs := dict -}} {{- range findRESubmatch `([a-z]+)="([^"]*)"` (index . 2) -}}{{- $attrs = merge $attrs (dict (index . 1) (index . 2)) -}}{{- end -}} {{- if eq (index . 1) "league" -}} {{- if $cur.id -}}{{- $leagues = $leagues | append $cur -}}{{- end -}} {{- $cur = dict "id" $attrs.id "title" $attrs.title "years" slice -}} {{- else if ne $attrs.index "false" -}} {{- $yid := $attrs.id | default (printf "%s-%s" $attrs.league $attrs.year) -}} {{- $text := $attrs.short | default ($attrs.label | default $attrs.year) -}} {{- $cur = merge $cur (dict "years" ($cur.years | append (dict "id" $yid "text" $text))) -}} {{- end -}} {{- end -}} {{- if $cur.id -}}{{- $leagues = $leagues | append $cur -}}{{- end -}}
