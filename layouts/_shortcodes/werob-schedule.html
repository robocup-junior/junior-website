{{- /* WEROB schedule. Cells are separated by " | ", {empty} is an empty cell and {colspan=3} at the start of a cell spans three columns. head="true": the first line holds the column headings (th scope="col"), the first cell of every other line is the row heading (th scope="row"), and on phones each row stacks. Optional: caption="...", labels="true" (show the column heading before each cell on phones). */ -}} {{- $page := .Page -}} {{- if eq (.Get "head") "true" -}} {{- $rows := slice -}} {{- range split (strings.TrimSpace .Inner) "\n" }}{{ with strings.TrimSpace . }}{{ $rows = $rows | append . }}{{ end }}{{ end -}} {{- $labels := slice -}} {{- with index $rows 0 }}{{ range split . " | " }}{{ $labels = $labels | append (strings.TrimSpace .) }}{{ end }}{{ end -}} {{- $caption := .Get "caption" -}}
{{- with $caption }}{{ end -}} {{ range $labels }}{{ end }} {{- range after 1 $rows -}} {{- range $i, $cell := split . " | " -}} {{- $cell = strings.TrimSpace $cell -}} {{- $span := hasPrefix $cell "{colspan=3}" -}} {{- if $span }}{{ $cell = strings.TrimPrefix "{colspan=3}" $cell }}{{ end -}} {{- if eq $cell "{empty}" }}{{ $cell = "" }}{{ end -}} {{- if eq $i 0 -}} {{- else -}} {{- $label := "" }}{{ if and (not $span) (lt $i (len $labels)) }}{{ $label = index $labels $i | plainify }}{{ end -}} {{ with $cell }}{{ $page.RenderString . }}{{ end }} {{- end -}} {{- end -}} {{- end -}}
{{ . }}
{{ if ne . "{empty}" }}{{ $page.RenderString . }}{{ end }}
{{ $page.RenderString $cell }}
{{- else -}} {{- if eq (.Get "striped") "true" }}{{ end }} {{- end -}}
